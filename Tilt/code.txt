
inductive Eq : (A : Type) -> (a : A) -> (b : A) -> Prop where
  | refl : (A : Type) -> (a : A) -> Eq A a a

inductive Nat : Type where
  | zero : Nat
  | succ : (n : Nat) -> Nat

inductive Int : Type where
  | ofNat : (n : Nat) -> Int
  | negSucc : (n : Nat) -> Int

#check Eq
#check Eq.refl
#check Eq.refl Nat

#check Nat
#check Nat.zero
#check Nat.succ

#check Int
#check Int.ofNat
#check Int.negSucc
#check Int.negSucc (Nat.succ Nat.zero)

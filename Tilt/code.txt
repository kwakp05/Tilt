inductive Or (a : Prop) (b : Prop) : Prop where
  | inl : (h : a) -> Or a b
  | inr : (h : b) -> Or a b

#check Or
#check Or.inl
#check Or.inr
#check Or.rec
def Or.elim : (a : Prop) -> (b : Prop) -> (c : Prop) -> (h : Or a b) -> (left : (_ : a) -> c) -> (right : (_ : b) -> c) -> c :=
  fun (a : Prop) =>
  fun (b : Prop) =>
  fun (c : Prop) =>
  fun (h : Or a b) =>
  fun (left : (_ : a) -> c) =>
  fun (right : (_ : b) -> c) =>
  Or.rec a b (fun (_ : Or a b) => c) left right h
#check Or.elim
#reduce Or.elim

inductive Int : Type where
  | ofNat : (n : Nat) -> Int
  | negSucc : (n : Nat) -> Int

def Int.natAbs : (m : Int) -> Nat := fun (m : Int) => Int.rec1 (fun (_ : Int) => Nat) (fun (x : Nat) => x) (fun (x : Nat) => Nat.succ x) m

#check Int
#check Int.ofNat
#check Int.negSucc
#check Int.negSucc (Nat.succ Nat.zero)
#check Int.natAbs
#reduce Int.natAbs (Int.ofNat (Nat.succ Nat.zero))
#reduce Int.natAbs (Int.negSucc (Nat.succ Nat.zero))

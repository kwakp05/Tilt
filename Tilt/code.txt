inductive False : Prop where

def Not : (a : Prop) -> Prop := fun (a : Prop) => (p : a) -> False

inductive Eq : (A : Type) -> (a : A) -> (b : A) -> Prop where
  | refl : (A : Type) -> (a : A) -> Eq A a a

inductive Bool : Type where
  | false : Bool
  | true : Bool

inductive Nat : Type where
  | zero : Nat
  | succ : (n : Nat) -> Nat

inductive Int : Type where
  | ofNat : (n : Nat) -> Int
  | negSucc : (n : Nat) -> Int

#check Not

#check Bool
